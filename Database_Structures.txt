*&---------------------------------------------------------------------*
*& View: ZV_TRANSLATION_STATUS
*& Purpose: View showing translation completeness status
*& Usage: Monitor translation progress across languages
*&---------------------------------------------------------------------*

VIEW ZV_TRANSLATION_STATUS:

SELECT 
  t.object_id,
  t.object_type,
  COUNT(*) as translated_languages,
  c.required_languages,
  CASE 
    WHEN COUNT(*) >= c.required_lang_count THEN 'COMPLETE'
    WHEN COUNT(*) > 0 THEN 'PARTIAL'
    ELSE 'MISSING'
  END as translation_status
FROM zcustom_translations t
LEFT JOIN (
  SELECT 
    object_type,
    config_value as required_languages,
    LENGTH(config_value) - LENGTH(REPLACE(config_value, ',', '')) + 1 as required_lang_count
  FROM zcustom_trans_config 
  WHERE config_key = 'REQUIRED_LANGUAGES'
) c ON t.object_type = c.object_type
GROUP BY t.object_id, t.object_type, c.required_languages, c.required_lang_count;
